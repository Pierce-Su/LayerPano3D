cmake_minimum_required(VERSION 3.20)
project(LayerPanoDepth LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# Build the original implementation, not an alternate depth algorithm.
get_filename_component(DEPTH_SRC "${CMAKE_CURRENT_LIST_DIR}/../../submodules/360monodepth/code/cpp" ABSOLUTE)
find_package(Python3 REQUIRED COMPONENTS Interpreter Development NumPy)
find_package(Eigen3 3.4.0 EXACT REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs highgui)
find_package(Ceres 2.1 REQUIRED)
find_package(Boost REQUIRED)
find_package(glog REQUIRED)
find_package(pybind11 REQUIRED)
find_package(OpenMP REQUIRED)
execute_process(COMMAND "${Python3_EXECUTABLE}" -c "import sysconfig; print(sysconfig.get_path('platlib'))" OUTPUT_VARIABLE SITE_PACKAGES OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
file(GLOB DEPTH_SOURCES "${DEPTH_SRC}/src/*.cpp")
list(REMOVE_ITEM DEPTH_SOURCES "${DEPTH_SRC}/src/EigenSolvers.cpp")
add_library(depth_stitch SHARED ${DEPTH_SOURCES})
target_include_directories(depth_stitch PRIVATE "${DEPTH_SRC}/include" "${DEPTH_SRC}/3rd_party" ${OpenCV_INCLUDE_DIRS} ${Boost_INCLUDE_DIRS})
target_link_libraries(depth_stitch PRIVATE Eigen3::Eigen Ceres::ceres ${OpenCV_LIBS} OpenMP::OpenMP_CXX)
Python3_add_library(depthmapAlign MODULE WITH_SOABI "${DEPTH_SRC}/python/instaOmniDepth/depthmapAlignModule.cpp")
target_include_directories(depthmapAlign PRIVATE "${DEPTH_SRC}/include" ${OpenCV_INCLUDE_DIRS})
target_link_libraries(depthmapAlign PRIVATE depth_stitch Python3::NumPy ${OpenCV_LIBS})
pybind11_add_module(EigenSolvers "${DEPTH_SRC}/src/EigenSolvers.cpp")
target_link_libraries(EigenSolvers PRIVATE Eigen3::Eigen glog::glog)
set_target_properties(depthmapAlign depth_stitch EigenSolvers PROPERTIES INSTALL_RPATH "$ORIGIN;${CMAKE_INSTALL_PREFIX}/lib")
install(TARGETS depth_stitch depthmapAlign LIBRARY DESTINATION "${SITE_PACKAGES}/instaOmniDepth")
install(TARGETS EigenSolvers LIBRARY DESTINATION "${SITE_PACKAGES}")
# Do not use upstream __init__: it overwrites LD_LIBRARY_PATH process-wide.
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/__init__.py" "\"\"\"Original 360monodepth native modules, user-prefix build.\"\"\"\n")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/__init__.py" DESTINATION "${SITE_PACKAGES}/instaOmniDepth")
